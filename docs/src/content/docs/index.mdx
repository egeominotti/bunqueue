```python title="jobs.py"
from bunqueue import Queue, Worker

connection = {"host": "127.0.0.1", "port": 6789}
queue = Queue("emails", **connection)
queue.add("welcome", {"to": "hello@example.com"})
queue.close()

def process(job):
    print("Processing:", job.data["to"])
    return {"sent": True}

Worker("emails", process, **connection).run()
```

```go title="main.go"
package main

import (
	"fmt"

	bunqueue "github.com/egeominotti/bunqueue/sdk/go"
)

func main() {
	queue := bunqueue.NewQueue("emails", bunqueue.Options{Host: "127.0.0.1", Port: 6789})
	defer queue.Close()
	if _, err := queue.Add("welcome", map[string]any{"to": "hello@example.com"}, nil); err != nil {
		panic(err)
	}

	worker := bunqueue.NewWorker("emails", func(job *bunqueue.Job) (any, error) {
		fmt.Println("Processing:", any(job.Data()).(map[string]any)["to"])
		return map[string]any{"sent": true}, nil
	}, bunqueue.WorkerOptions{Host: "127.0.0.1", Port: 6789})
	worker.Run()
}
```

```bash
go run .
```

```rust title="src/main.rs"
use bunqueue_client::{ConnectionOptions, JobOptions, Queue, Value, Worker, WorkerOptions};

fn main() -> Result<(), Box<dyn std::error::Error>> {
    let connection = ConnectionOptions {
        host: "127.0.0.1".into(),
        port: 6789,
        ..Default::default()
    };

    let queue = Queue::new("emails", connection.clone());
    let data = Value::Map(vec![(Value::from("to"), Value::from("hello@example.com"))]);
    queue.add("welcome", data, JobOptions::default())?;

    let worker = Worker::new(
        "emails",
        |job| {
            println!("Processing: {}", job.data()["to"].as_str().unwrap_or_default());
            Ok(Value::Map(vec![(Value::from("sent"), Value::from(true))]))
        },
        WorkerOptions { connection, ..Default::default() },
    );
    worker.run()?;
    Ok(())
}
```

```bash
cargo run
```
